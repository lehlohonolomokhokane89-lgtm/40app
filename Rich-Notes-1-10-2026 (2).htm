create table if not exists drivers (
id uuid default gen_random_uuid() primary key,
plate text unique,
taxi_plate text,
driver_name text,
driver_phone text,
route text,
from_town text,
to_town text,
is_live boolean default false,
is_gold boolean default false,
is_verified boolean default false,
last_seen timestamp default now()
);
alter table drivers enable row level security;
drop policy if exists "Anyone can read live" on drivers;
drop policy if exists "Anyone can go live" on drivers;
drop policy if exists "Anyone can update live" on drivers;
create policy "Anyone can read live" on drivers for select using (true);
create policy "Anyone can go live" on drivers for insert with check (true);
create policy "Anyone can update live" on drivers for update using (true);
